cmake_minimum_required(VERSION 3.20)
project(saga VERSION 0.1.0 LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 23)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
option(SAGA_SANITIZERS "Enable address and undefined behavior sanitizers" OFF)
find_package(Threads REQUIRED)
find_package(SQLite3 QUIET)
find_package(CURL QUIET)
# Vendored Lexbor 3.0.0, Apache-2.0. Only HTML/DOM/encoding/Unicode modules.
file(GLOB_RECURSE SAGA_LEXBOR_SOURCES CONFIGURE_DEPENDS
  third_party/lexbor/source/lexbor/core/*.c third_party/lexbor/source/lexbor/dom/*.c
  third_party/lexbor/source/lexbor/html/*.c third_party/lexbor/source/lexbor/encoding/*.c
  third_party/lexbor/source/lexbor/ns/*.c third_party/lexbor/source/lexbor/tag/*.c
  third_party/lexbor/source/lexbor/unicode/*.c third_party/lexbor/source/lexbor/punycode/*.c third_party/lexbor/source/lexbor/ports/posix/*.c)
add_library(saga_lexbor STATIC ${SAGA_LEXBOR_SOURCES})
target_include_directories(saga_lexbor SYSTEM PUBLIC third_party/lexbor/source)
target_link_libraries(saga_lexbor PUBLIC m)
if(NOT SQLite3_FOUND)
  find_library(SAGA_SQLITE_LIBRARY NAMES sqlite3 libsqlite3.so.0 REQUIRED)
  add_library(SQLite::SQLite3 UNKNOWN IMPORTED)
  set_target_properties(SQLite::SQLite3 PROPERTIES IMPORTED_LOCATION "${SAGA_SQLITE_LIBRARY}" INTERFACE_INCLUDE_DIRECTORIES "${CMAKE_CURRENT_SOURCE_DIR}/third_party/compat")
endif()
if(NOT CURL_FOUND)
  find_library(SAGA_CURL_LIBRARY NAMES curl libcurl.so.4 REQUIRED)
  add_library(CURL::libcurl UNKNOWN IMPORTED)
  set_target_properties(CURL::libcurl PROPERTIES IMPORTED_LOCATION "${SAGA_CURL_LIBRARY}" INTERFACE_INCLUDE_DIRECTORIES "${CMAKE_CURRENT_SOURCE_DIR}/third_party/compat")
endif()
set(CURSES_NEED_WIDE TRUE)
find_package(Curses QUIET)
if(NOT CURSES_FOUND)
  find_library(SAGA_CURSES_LIBRARY NAMES ncursesw libncursesw.so.6 REQUIRED)
  set(CURSES_LIBRARIES "${SAGA_CURSES_LIBRARY}")
  set(CURSES_INCLUDE_DIRS "${CMAKE_CURRENT_SOURCE_DIR}/third_party/compat")
endif()
set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "${CMAKE_CURRENT_SOURCE_DIR}/src/schema.sql")
file(READ "${CMAKE_CURRENT_SOURCE_DIR}/src/schema.sql" SAGA_SCHEMA)
configure_file(cmake/schema.hpp.in generated/schema.hpp @ONLY)
find_package(Git QUIET)
set(SAGA_GIT_SHA "unknown")
set(SAGA_GIT_BRANCH "unknown")
set(SAGA_GIT_DIRTY "unknown")
if(Git_FOUND)
  execute_process(COMMAND "${GIT_EXECUTABLE}" rev-parse HEAD WORKING_DIRECTORY "${CMAKE_CURRENT_SOURCE_DIR}" OUTPUT_VARIABLE SAGA_GIT_SHA OUTPUT_STRIP_TRAILING_WHITESPACE ERROR_QUIET)
  execute_process(COMMAND "${GIT_EXECUTABLE}" branch --show-current WORKING_DIRECTORY "${CMAKE_CURRENT_SOURCE_DIR}" OUTPUT_VARIABLE SAGA_GIT_BRANCH OUTPUT_STRIP_TRAILING_WHITESPACE ERROR_QUIET)
  execute_process(COMMAND "${GIT_EXECUTABLE}" status --porcelain WORKING_DIRECTORY "${CMAKE_CURRENT_SOURCE_DIR}" OUTPUT_VARIABLE SAGA_GIT_STATUS ERROR_QUIET)
  if(SAGA_GIT_STATUS)
    set(SAGA_GIT_DIRTY "true")
  else()
    set(SAGA_GIT_DIRTY "false")
  endif()
endif()
string(TIMESTAMP SAGA_BUILD_TIME "%Y-%m-%dT%H:%M:%SZ" UTC)
string(TOUPPER "${CMAKE_BUILD_TYPE}" SAGA_TYPE)
set(SAGA_TYPE_FLAGS "${CMAKE_CXX_FLAGS_${SAGA_TYPE}}")
configure_file(cmake/build_metadata.hpp.in generated/build_metadata.hpp @ONLY)
add_library(saga_core src/debug.cpp src/common.cpp src/db.cpp src/persona.cpp src/model.cpp src/generation.cpp src/memory.cpp src/tools.cpp src/runtime.cpp src/ipc.cpp src/edit.cpp src/web.cpp src/research.cpp src/search/search.cpp src/search/duckduckgo.cpp src/search/fourget.cpp src/web/document.cpp src/web/entities.c src/web/html.cpp src/web/retrieval.cpp src/web/detect.cpp src/web/sources.cpp src/web/acquisition.cpp)
target_include_directories(saga_core PUBLIC include)
target_include_directories(saga_core SYSTEM PUBLIC third_party)
target_include_directories(saga_core PRIVATE "${CMAKE_CURRENT_BINARY_DIR}/generated")
target_link_libraries(saga_core PUBLIC SQLite::SQLite3 CURL::libcurl Threads::Threads saga_lexbor)
target_compile_options(saga_core PRIVATE -Wall -Wextra -Wpedantic)
target_precompile_headers(saga_core PRIVATE "$<$<COMPILE_LANGUAGE:CXX>:${CMAKE_CURRENT_SOURCE_DIR}/include/saga/common.hpp>")
if(SAGA_SANITIZERS)
  include(CheckCXXSourceCompiles)
  set(SAGA_PREVIOUS_REQUIRED_FLAGS "${CMAKE_REQUIRED_FLAGS}")
  set(CMAKE_REQUIRED_FLAGS "${CMAKE_REQUIRED_FLAGS} -fsanitize=address,undefined")
  check_cxx_source_compiles("int main() { return 0; }" SAGA_SANITIZER_LINKS)
  set(CMAKE_REQUIRED_FLAGS "${SAGA_PREVIOUS_REQUIRED_FLAGS}")
  if(NOT SAGA_SANITIZER_LINKS)
    message(FATAL_ERROR "SAGA_SANITIZERS requires the compiler's AddressSanitizer and UBSan runtime/development libraries.")
  endif()
  target_compile_options(saga_core PUBLIC -fsanitize=address,undefined -fno-omit-frame-pointer)
  target_compile_options(saga_lexbor PRIVATE -fsanitize=address,undefined -fno-omit-frame-pointer)
  target_link_options(saga_core PUBLIC -fsanitize=address,undefined)
endif()
add_executable(sagad src/daemon.cpp)
add_executable(saga src/client.cpp)
find_library(SAGA_MARKDOWN_LIBRARY NAMES md4c libmd4c.so.0 REQUIRED)
find_path(SAGA_MARKDOWN_INCLUDE md4c.h)
if(NOT SAGA_MARKDOWN_INCLUDE)
  set(SAGA_MARKDOWN_INCLUDE "${CMAKE_CURRENT_SOURCE_DIR}/third_party/compat")
endif()
target_link_libraries(saga_core PUBLIC ${SAGA_MARKDOWN_LIBRARY})
target_include_directories(saga_core SYSTEM PRIVATE ${SAGA_MARKDOWN_INCLUDE})
add_library(saga_chat_ui src/chat_ui.cpp src/markdown.cpp)
target_link_libraries(saga_chat_ui PUBLIC saga_core PRIVATE ${CURSES_LIBRARIES} ${SAGA_MARKDOWN_LIBRARY})
target_include_directories(saga_chat_ui SYSTEM PRIVATE ${SAGA_MARKDOWN_INCLUDE})
target_include_directories(saga_chat_ui SYSTEM PRIVATE ${CURSES_INCLUDE_DIRS})
target_compile_options(saga_chat_ui PRIVATE -Wall -Wextra -Wpedantic)
target_link_libraries(saga PRIVATE saga_chat_ui)
foreach(target saga sagad)
  target_link_libraries(${target} PRIVATE saga_core)
  target_compile_options(${target} PRIVATE -Wall -Wextra -Wpedantic)
endforeach()
