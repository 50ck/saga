cmake_minimum_required(VERSION 3.20)
project(saga VERSION 0.1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 23)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
option(SAGA_SANITIZERS "Enable address and undefined behavior sanitizers" OFF)
find_package(Threads REQUIRED)
find_package(SQLite3 QUIET)
find_package(CURL QUIET)
if(NOT SQLite3_FOUND)
  find_library(SAGA_SQLITE_LIBRARY NAMES sqlite3 libsqlite3.so.0 REQUIRED)
  add_library(SQLite::SQLite3 UNKNOWN IMPORTED)
  set_target_properties(SQLite::SQLite3 PROPERTIES IMPORTED_LOCATION "${SAGA_SQLITE_LIBRARY}" INTERFACE_INCLUDE_DIRECTORIES "${CMAKE_CURRENT_SOURCE_DIR}/third_party/compat")
endif()
if(NOT CURL_FOUND)
  find_library(SAGA_CURL_LIBRARY NAMES curl libcurl.so.4 REQUIRED)
  add_library(CURL::libcurl UNKNOWN IMPORTED)
  set_target_properties(CURL::libcurl PROPERTIES IMPORTED_LOCATION "${SAGA_CURL_LIBRARY}" INTERFACE_INCLUDE_DIRECTORIES "${CMAKE_CURRENT_SOURCE_DIR}/third_party/compat")
endif()
set(CURSES_NEED_WIDE TRUE)
find_package(Curses QUIET)
if(NOT CURSES_FOUND)
  find_library(SAGA_CURSES_LIBRARY NAMES ncursesw libncursesw.so.6 REQUIRED)
  set(CURSES_LIBRARIES "${SAGA_CURSES_LIBRARY}")
  set(CURSES_INCLUDE_DIRS "${CMAKE_CURRENT_SOURCE_DIR}/third_party/compat")
endif()
set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "${CMAKE_CURRENT_SOURCE_DIR}/src/schema.sql")
file(READ "${CMAKE_CURRENT_SOURCE_DIR}/src/schema.sql" SAGA_SCHEMA)
configure_file(cmake/schema.hpp.in generated/schema.hpp @ONLY)
add_library(saga_core src/common.cpp src/db.cpp src/persona.cpp src/model.cpp src/memory.cpp src/tools.cpp src/runtime.cpp src/ipc.cpp src/edit.cpp)
target_include_directories(saga_core PUBLIC include)
target_include_directories(saga_core SYSTEM PUBLIC third_party)
target_include_directories(saga_core PRIVATE "${CMAKE_CURRENT_BINARY_DIR}/generated")
target_link_libraries(saga_core PUBLIC SQLite::SQLite3 CURL::libcurl Threads::Threads)
target_compile_options(saga_core PRIVATE -Wall -Wextra -Wpedantic)
target_precompile_headers(saga_core PRIVATE <saga/common.hpp>)
if(SAGA_SANITIZERS)
  include(CheckCXXSourceCompiles)
  set(SAGA_PREVIOUS_REQUIRED_FLAGS "${CMAKE_REQUIRED_FLAGS}")
  set(CMAKE_REQUIRED_FLAGS "${CMAKE_REQUIRED_FLAGS} -fsanitize=address,undefined")
  check_cxx_source_compiles("int main() { return 0; }" SAGA_SANITIZER_LINKS)
  set(CMAKE_REQUIRED_FLAGS "${SAGA_PREVIOUS_REQUIRED_FLAGS}")
  if(NOT SAGA_SANITIZER_LINKS)
    message(FATAL_ERROR "SAGA_SANITIZERS requires the compiler's AddressSanitizer and UBSan runtime/development libraries.")
  endif()
  target_compile_options(saga_core PUBLIC -fsanitize=address,undefined -fno-omit-frame-pointer)
  target_link_options(saga_core PUBLIC -fsanitize=address,undefined)
endif()
add_executable(sagad src/daemon.cpp)
add_executable(saga src/client.cpp)
find_library(SAGA_MARKDOWN_LIBRARY NAMES md4c libmd4c.so.0 REQUIRED)
find_path(SAGA_MARKDOWN_INCLUDE md4c.h)
if(NOT SAGA_MARKDOWN_INCLUDE)
  set(SAGA_MARKDOWN_INCLUDE "${CMAKE_CURRENT_SOURCE_DIR}/third_party/compat")
endif()
add_library(saga_chat_ui src/chat_ui.cpp src/markdown.cpp)
target_link_libraries(saga_chat_ui PUBLIC saga_core PRIVATE ${CURSES_LIBRARIES} ${SAGA_MARKDOWN_LIBRARY})
target_include_directories(saga_chat_ui SYSTEM PRIVATE ${SAGA_MARKDOWN_INCLUDE})
target_include_directories(saga_chat_ui SYSTEM PRIVATE ${CURSES_INCLUDE_DIRS})
target_compile_options(saga_chat_ui PRIVATE -Wall -Wextra -Wpedantic)
target_link_libraries(saga PRIVATE saga_chat_ui)
foreach(target saga sagad)
  target_link_libraries(${target} PRIVATE saga_core)
  target_compile_options(${target} PRIVATE -Wall -Wextra -Wpedantic)
endforeach()
include(CTest)
if(BUILD_TESTING)
  add_executable(saga_chat_ui_tests tests/chat_ui_tests.cpp)
  target_link_libraries(saga_chat_ui_tests PRIVATE saga_chat_ui)
  target_compile_definitions(saga_chat_ui_tests PRIVATE SAGA_MARKDOWN_FIXTURE="${CMAKE_CURRENT_SOURCE_DIR}/tests/markdown.md")
  target_compile_options(saga_chat_ui_tests PRIVATE -Wall -Wextra -Wpedantic)
  add_test(NAME saga_chat_ui COMMAND saga_chat_ui_tests)
  set_tests_properties(saga_chat_ui PROPERTIES TIMEOUT 60)
  add_executable(saga_tests tests/tests.cpp)
  target_link_libraries(saga_tests PRIVATE saga_core)
  target_compile_options(saga_tests PRIVATE -Wall -Wextra -Wpedantic)
  add_test(NAME saga_core COMMAND saga_tests)
  add_executable(saga_integration tests/integration.cpp)
  target_link_libraries(saga_integration PRIVATE saga_chat_ui)
  target_compile_options(saga_integration PRIVATE -Wall -Wextra -Wpedantic)
  add_test(NAME saga_end_to_end COMMAND saga_integration $<TARGET_FILE:sagad> $<TARGET_FILE:saga>)
  set_tests_properties(saga_end_to_end PROPERTIES SKIP_RETURN_CODE 77)
  add_executable(saga_runtime_tests tests/runtime_tests.cpp)
  target_link_libraries(saga_runtime_tests PRIVATE saga_core)
  target_compile_options(saga_runtime_tests PRIVATE -Wall -Wextra -Wpedantic)
  add_test(NAME saga_runtime_lifecycle COMMAND saga_runtime_tests)
  add_executable(saga_daemon_protocol_tests tests/daemon_protocol.cpp)
  target_link_libraries(saga_daemon_protocol_tests PRIVATE saga_core)
  target_compile_options(saga_daemon_protocol_tests PRIVATE -Wall -Wextra -Wpedantic)
  add_test(NAME saga_daemon_protocol COMMAND saga_daemon_protocol_tests)
  set_tests_properties(saga_runtime_lifecycle PROPERTIES TIMEOUT 60)
  set_tests_properties(saga_core saga_end_to_end saga_daemon_protocol PROPERTIES TIMEOUT 60)
endif()
install(TARGETS saga sagad RUNTIME DESTINATION bin)
